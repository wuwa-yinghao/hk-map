    <!-- 6. 重置與生成按鈕 -->
    <button type="button" id="reset-btn" class="bottom-screenshot-btn" aria-label="重置全部欄位" data-html2canvas-ignore="true" style="border-color: rgba(255,92,92,0.35); color: var(--neg);">
      <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
        <polyline points="1 4 1 10 7 10"></polyline>
        <path d="M3.51 15a9 9 0 1 0 2.13-9.36L1 10"></path>

      <span>重置全部欄位</span>

    <button type="button" id="screenshot-btn" class="bottom-screenshot-btn" aria-label="生成截圖" data-html2canvas-ignore="true">
      <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
        <path d="M23 19a2 2 0 0 1-2 2H3a2 2 0 0 1-2 2V8a2 2 0 0 1 2-2h4l2-3h6l2 3h4a2 2 0 0 1 2 2z"></path>
        <circle cx="12" cy="13" r="4"></circle>
      </svg>
      <span>生成截圖</span>

<script>
  const STORAGE_KEY = 'ud_calculator_state_v1';
  const allFieldIds = ['src-amount', 'src-rate', 'up-amount', 'up-point', 'up-rate', 'down-amount', 'down-point', 'down-rate'];

  function stripCommas(str) {
    return (str || '').toString().replace(/,/g, '');
  }
  function formatThousands(numStr) {
    const cleaned = stripCommas(numStr);
    if (cleaned === '' || cleaned === '-' || isNaN(parseFloat(cleaned))) return cleaned;
    return parseFloat(cleaned).toLocaleString('en-US', { maximumFractionDigits: 6 });
  }

  const inputs = [
    'src-amount', 'src-rate',
    'up-amount', 'up-point', 'up-rate',
    'down-amount', 'down-point', 'down-rate'
  ].map(id => document.getElementById(id));

  function setAppHeight() {
    const h = (window.visualViewport && window.visualViewport.height) || window.innerHeight;
    document.documentElement.style.setProperty('--app-height', h + 'px');
  }
  setAppHeight();
  window.addEventListener('resize', setAppHeight);

  let mode = 'deposit'; 
  const btnDeposit = document.getElementById('btn-deposit');
  const btnWithdraw = document.getElementById('btn-withdraw');
  const dashboardTitle = document.getElementById('dashboard-title');

  function setMode(newMode) {
    mode = newMode;
    btnDeposit.classList.toggle('active', mode === 'deposit');
    btnWithdraw.classList.toggle('active', mode === 'withdraw');
    dashboardTitle.textContent = `上下游利潤分析 (${mode === 'deposit' ? '入金' : '出金'})`;
    calculate();
  }
  btnDeposit.addEventListener('click', () => setMode('deposit'));
  btnWithdraw.addEventListener('click', () => setMode('withdraw'));

  const accordion = document.getElementById('market-accordion');
  document.getElementById('accordion-toggle').addEventListener('click', () => {
    accordion.classList.toggle('open');
  });

  function calcLeg(amount, pointInput, rate) {
    const point = (100 - pointInput) / 100;
    return (amount * point) / rate;
  }

  function updateDiffDisplay(diffAbsolute, diffPercent, absEl, pctEl, fillEl, markerEl) {
    absEl.textContent = (diffAbsolute >= 0 ? '+' : '') + diffAbsolute.toFixed(3);
    pctEl.textContent = (diffPercent >= 0 ? '+' : '') + diffPercent.toFixed(2) + '%';

    absEl.classList.remove('text-pos','text-neg','text-zero');
    pctEl.classList.remove('text-pos','text-neg','text-zero');

    let cls = 'text-zero';
    if (diffAbsolute > 0) cls = 'text-pos';
    else if (diffAbsolute < 0) cls = 'text-neg';
    absEl.classList.add(cls);
    pctEl.classList.add(cls);

    const clamped = Math.max(-50, Math.min(50, diffPercent));
    const travel = (clamped / 50) * 50;

    if (diffAbsolute >= 0) {
      fillEl.style.left = '50%';
      fillEl.style.width = Math.abs(travel) + '%';
      fillEl.style.background = 'linear-gradient(90deg, rgba(47,209,128,0.15), #2FD180)';
    } else {
      fillEl.style.left = (50 - Math.abs(travel)) + '%';
      fillEl.style.width = Math.abs(travel) + '%';
      fillEl.style.background = 'linear-gradient(90deg, #FF5C5C, rgba(255,92,92,0.15))';
    }
    markerEl.style.left = (50 + travel) + '%';
    markerEl.style.borderColor = diffAbsolute > 0 ? '#2FD180' : diffAbsolute < 0 ? '#FF5C5C' : '#565D70';
  }

  const diffAbsEl = document.getElementById('diff-absolute');
  const diffPercentEl = document.getElementById('diff-percent');
  const fill = document.getElementById('gauge-fill');
  const marker = document.getElementById('gauge-marker');

  const srcDiffAbsEl = document.getElementById('src-diff-absolute');
  const srcDiffPercentEl = document.getElementById('src-diff-percent');
  const srcFill = document.getElementById('src-gauge-fill');
  const srcMarker = document.getElementById('src-gauge-marker');

  function calculate() {
    const srcAmount = parseFloat(stripCommas(document.getElementById('src-amount').value)) || 0;
    const srcRate = parseFloat(document.getElementById('src-rate').value) || 1;
    const srcResult = srcAmount / srcRate;
    document.getElementById('src-result').textContent = srcResult.toFixed(3);

    const upAmount = parseFloat(stripCommas(document.getElementById('up-amount').value)) || 0;
    const upPointInput = parseFloat(document.getElementById('up-point').value) || 0;
    const upRate = parseFloat(document.getElementById('up-rate').value) || 1;
    const upResult = calcLeg(upAmount, upPointInput, upRate);

    const downAmount = parseFloat(stripCommas(document.getElementById('down-amount').value)) || 0;
    const downPointInput = parseFloat(document.getElementById('down-point').value) || 0;
    const downRate = parseFloat(document.getElementById('down-rate').value) || 1;
    const downResult = calcLeg(downAmount, downPointInput, downRate);

    document.getElementById('up-result').textContent = upResult.toFixed(3);
    document.getElementById('down-result').textContent = downResult.toFixed(3);

    const diffAbsolute = mode === 'deposit' ? (upResult - downResult) : (downResult - upResult);
    const base = mode === 'deposit' ? upResult : downResult;
    const diffPercent = base !== 0 ? (diffAbsolute / base) * 100 : 0;
    updateDiffDisplay(diffAbsolute, diffPercent, diffAbsEl, diffPercentEl, fill, marker);

    const srcDiffAbsolute = mode === 'deposit' ? (srcResult - upResult) : (upResult - srcResult);
    const srcBase = mode === 'deposit' ? srcResult : upResult;
    const srcDiffPercent = srcBase !== 0 ? (srcDiffAbsolute / srcBase) * 100 : 0;
    updateDiffDisplay(srcDiffAbsolute, srcDiffPercent, srcDiffAbsEl, srcDiffPercentEl, srcFill, srcMarker);

    saveState();
  }

  inputs.forEach(input => input.addEventListener('input', calculate));

  const srcAmountEl = document.getElementById('src-amount');
  const upAmountEl = document.getElementById('up-amount');
  const downAmountEl = document.getElementById('down-amount');
  const amountEls = [srcAmountEl, upAmountEl, downAmountEl];

  amountEls.forEach(el => {
    el.addEventListener('input', () => {
      const raw = stripCommas(el.value);
      amountEls.forEach(other => { if (other !== el) other.value = raw; });
      calculate();
    });
    el.addEventListener('focus', () => {
      el.value = stripCommas(el.value);
      setTimeout(() => el.select(), 0);
    });
    el.addEventListener('blur', () => {
      const formatted = formatThousands(el.value);
      amountEls.forEach(other => { other.value = formatted; });
    });
  });

  // 非金額欄位：聚焦時自動全選
  document.querySelectorAll('input').forEach(input => {
    if (!amountEls.includes(input)) {
      input.addEventListener('focus', () => {
        setTimeout(() => input.select(), 0);
      });
    }
  });

  document.querySelectorAll('.sign-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const targetInput = document.getElementById(btn.dataset.target);
      if (!targetInput) return;
      targetInput.value = (parseFloat(targetInput.value) || 0) * -1;
      calculate();
    });
  });

  document.querySelectorAll('.qa-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const targetInput = document.getElementById(btn.dataset.target);
      if (!targetInput) return;
      const current = parseFloat(targetInput.value) || 0;
      let next = btn.dataset.percent !== undefined 
        ? current * (1 + parseFloat(btn.dataset.percent) / 100) 
        : current + parseFloat(btn.dataset.delta);
      targetInput.value = Math.round(next * 10000) / 10000;
      calculate();
    });
  });

  document.querySelectorAll('input').forEach(input => {
    input.addEventListener('keydown', (e) => { if (e.key === 'Enter') input.blur(); });
  });

  // ==========================================
  // 記住上次輸入（localStorage）
  // ==========================================
  function saveState() {
    try {
      const state = {};
      allFieldIds.forEach(id => {
        const el = document.getElementById(id);
        state[id] = amountEls.includes(el) ? stripCommas(el.value) : el.value;
      });
      state.mode = mode;
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
    } catch (e) {}
  }

  function loadState() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return null;
      return JSON.parse(raw);
    } catch (e) { return null; }
  }

  // ==========================================
  // 一鍵重置
  // ==========================================
  const resetBtn = document.getElementById('reset-btn');
  resetBtn.addEventListener('click', () => {
    document.getElementById('src-amount').value = '0';
    document.getElementById('src-rate').value = '1';
    document.getElementById('up-amount').value = '0';
    document.getElementById('up-point').value = '0';
    document.getElementById('up-rate').value = '1';
    document.getElementById('down-amount').value = '0';
    document.getElementById('down-point').value = '0';
    document.getElementById('down-rate').value = '1';
    amountEls.forEach(el => { el.value = formatThousands(el.value); });
    try { localStorage.removeItem(STORAGE_KEY); } catch (e) {}
    calculate();
    showToast('已重置所有欄位');
  });

  // ==========================================
  // 初始化：套用上次保存的狀態
  // ==========================================
  const savedState = loadState();
  if (savedState) {
    allFieldIds.forEach(id => {
      if (savedState[id] !== undefined) document.getElementById(id).value = savedState[id];
    });
  }
  amountEls.forEach(el => { el.value = formatThousands(el.value); });
  if (savedState && savedState.mode) {
    setMode(savedState.mode);
  } else {
    calculate();
  }

  // ==========================================
  // 超採樣截圖引擎與精準原生分享/長按控制
  // ==========================================
  const screenshotBtn = document.getElementById('screenshot-btn');
  const toast = document.getElementById('toast');
  const previewModal = document.getElementById('preview-modal');
  const previewImg = document.getElementById('preview-img');
  const closeModal = document.getElementById('close-modal');
  const copyImgBtn = document.getElementById('copy-img-btn');
  const saveImgBtn = document.getElementById('save-img-btn');
  const dynamicTip = document.getElementById('dynamic-tip');

  function showToast(message) {
    toast.textContent = message; toast.classList.add('show');
    setTimeout(() => { toast.classList.remove('show'); }, 3000);
  }

  // 檢查當前環境是否能直接喚起原生 Share Sheet
  function checkShareSupport() {
    try {
      if (navigator.share && navigator.canShare) {
        const mockFile = new File([new Uint8Array([137,80,78,71])], 't.png', { type: 'image/png' });
        return navigator.canShare({ files: [mockFile] });
      }
    } catch(e) {}
    return false;
  }

  closeModal.addEventListener('click', () => { previewModal.classList.remove('show'); });

  screenshotBtn.addEventListener('click', async () => {
    if (screenshotBtn.disabled) return;
    screenshotBtn.disabled = true;
    showToast('正在進行超採樣渲染...');

    const currentScrollY = window.scrollY;

    try {
      window.scrollTo(0, 0);
      await new Promise(resolve => setTimeout(resolve, 120));
      
      const container = document.querySelector('.container');
      const width = container.offsetWidth;
      const height = container.offsetHeight;

      const canvas = await html2canvas(container, {
        backgroundColor: '#0A0C10',
        scale: 4,                  
        useCORS: true,             
        logging: false,
        allowTaint: false,
        width: width,
        height: height,
        scrollX: 0,
        scrollY: 0,
        onclone: (clonedDoc) => {
          const clonedInputs = clonedDoc.querySelectorAll('input');
          clonedInputs.forEach(input => {
            const val = input.value || '';
            const parent = input.parentNode;
            const fakeSpan = clonedDoc.createElement('span');
            fakeSpan.innerText = val;
            fakeSpan.className = 'screenshot-text-node';
            parent.replaceChild(fakeSpan, input);
          });
        }
      });

      const dataUrl = canvas.toDataURL('image/png', 1.0);
      previewImg.src = dataUrl;

      if (checkShareSupport()) {
        dynamicTip.innerHTML = "💡 點擊下方<b>「保存到相冊」</b>可直接喚起原生分享選單！";
      } else {
        dynamicTip.innerHTML = "💡 當前 App 內建瀏覽器限制，請<b>「直接長按下方圖片」</b>即可彈出儲存/分享選單！";
      }

      previewModal.classList.add('show');
      showToast('無損截圖已完成！');

    } catch (error) {
      console.error(error);
      showToast('截圖失敗，請使用手機系統內建截圖！');
    } finally {
      window.scrollTo(0, currentScrollY);
      screenshotBtn.disabled = false;
    }
  });

  // 功能 1：複製圖片至剪貼簿
  copyImgBtn.addEventListener('click', async () => {
    if (!previewImg.src) return;
    try {
      const response = await fetch(previewImg.src);
      const blob = await response.blob();
      await navigator.clipboard.write([
        new ClipboardItem({ [blob.type]: blob })
      ]);
      showToast('✅ 圖片已成功複製到剪貼簿！');
    } catch (error) {
      console.error(error);
      showToast('❌ 複製失敗，請長按圖片進行儲存。');
    }
  });

  // 功能 2：極致防黑畫面的保存控制
  saveImgBtn.addEventListener('click', async () => {
    if (!previewImg.src) return;
    try {
      const d = new Date();
      const dateString = `${d.getFullYear()}${(d.getMonth()+1).toString().padStart(2,'0')}${d.getDate().toString().padStart(2,'0')}_${d.getHours().toString().padStart(2,'0')}${d.getMinutes().toString().padStart(2,'0')}`;
      const fileName = `上下游利潤分析_${dateString}.png`;

      const response = await fetch(previewImg.src);
      const blob = await response.blob();
      const file = new File([blob], fileName, { type: 'image/png' });

      if (navigator.share && navigator.canShare && navigator.canShare({ files: [file] })) {
        await navigator.share({
          files: [file],
          title: '利益分析圖'
        });
      } else {
        showToast('👉 當前環境限制，請直接「長按圖片」即可儲存與分享！');
        previewImg.scrollIntoView({ behavior: 'smooth', block: 'center' });
        
        previewImg.style.transform = 'scale(1.04)';
        setTimeout(() => { previewImg.style.transform = 'scale(1)'; }, 250);
      }
    } catch (error) {
      if (error.name !== 'AbortError') {
        showToast('💡 請直接「長按上方圖片」進行儲存或分享！');
      }
    }
  });
</script>
